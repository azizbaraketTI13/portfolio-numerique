Programme 1 — Addition

#include <stdio.h>

int main(void) {
    int a = 5, b = 3;
    printf("Somme : %d\n", a + b);
    return 0;
}

Ce programme calcule et affiche la somme de deux entiers (a + b)

Programme 2 — Pair ou impair

#include <stdio.h>

int main(void) {
    int n = 8;

    if (n % 2 == 0)
        printf("Pair\n");
    else
        printf("Impair\n");

    return 0;
}

Ce programme vérifie si le nombre n est pair ou impair

Programme 3 — Boucle

#include <stdio.h>

int main(void) {
    for (int i = 1; i <= 5; i++)
        printf("%d\n", i);

    return 0;
}

Ce programme affiche les nombres de 1 à 5
